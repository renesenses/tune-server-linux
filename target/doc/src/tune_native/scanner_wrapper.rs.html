13#[pyfunction]
14fn list_audio_files(py: Python<'_>, dirs: Vec<String>) -> PyResult<pyo3::Bound<'_, PyList>> {
15    let files = py.allow_threads(|| walker::list_audio_files(&dirs));
16    let list = PyList::empty(py);
17    for f in files {
18        list.append(f.to_string_lossy().to_string())?;
19    }
20    Ok(list)
21}

23#[pyfunction]
24#[pyo3(signature = (dirs, with_hash=false))]
25fn scan_directories<'py>(
26    py: Python<'py>,
27    dirs: Vec<String>,
28    with_hash: bool,
29) -> PyResult<pyo3::Bound<'py, PyDict>> {
30    let (results, stats) = py.allow_threads(|| {
31        walker::scan_directories(&dirs, with_hash, None)
32    });
33
34    let dict = PyDict::new(py);
35    let files_list = PyList::empty(py);
36
37    for f in &results {
38        let file_dict = PyDict::new(py);
39        file_dict.set_item("path", &f.path)?;
40        file_dict.set_item("file_size", f.file_size)?;
41        file_dict.set_item("mtime", f.mtime)?;
42
43        if let Some(ref meta) = f.metadata {
44            let meta_dict = PyDict::new(py);
45            macro_rules! set_opt {
46                ($key:expr, $val:expr) => {
47                    if let Some(ref v) = $val { meta_dict.set_item($key, v)?; }
48                };
49            }
50            set_opt!("title", meta.title);
51            set_opt!("artist", meta.artist);
52            set_opt!("album", meta.album);
53            set_opt!("album_artist", meta.album_artist);
54            set_opt!("album_artist_sort", meta.album_artist_sort);
55            set_opt!("track_number", meta.track_number);
56            set_opt!("disc_number", meta.disc_number);
57            set_opt!("disc_subtitle", meta.disc_subtitle);
58            set_opt!("year", meta.year);
59            set_opt!("original_year", meta.original_year);
60            set_opt!("genre", meta.genre);
61            set_opt!("duration_ms", meta.duration_ms);
62            set_opt!("sample_rate", meta.sample_rate);
63            set_opt!("bit_depth", meta.bit_depth);
64            set_opt!("channels", meta.channels);
65            set_opt!("format", meta.format);
66            set_opt!("file_size", meta.file_size);
67            set_opt!("bpm", meta.bpm);
68            set_opt!("label", meta.label);
69            set_opt!("catalog_number", meta.catalog_number);
70            set_opt!("release_date", meta.release_date);
71            set_opt!("original_date", meta.original_date);
72            set_opt!("musicbrainz_recording_id", meta.musicbrainz_recording_id);
73            set_opt!("musicbrainz_release_id", meta.musicbrainz_release_id);
74            set_opt!("musicbrainz_artist_id", meta.musicbrainz_artist_id);
75            set_opt!("musicbrainz_album_artist_id", meta.musicbrainz_album_artist_id);
76            set_opt!("musicbrainz_release_group_id", meta.musicbrainz_release_group_id);
77            set_opt!("isrc", meta.isrc);
78            meta_dict.set_item("has_cover", meta.has_cover)?;
79            meta_dict.set_item("compilation", meta.compilation)?;
80            file_dict.set_item("metadata", meta_dict)?;
81        }
82
83        if let Some(ref hash) = f.audio_hash {
84            file_dict.set_item("audio_hash", hash)?;
85        }
86
87        files_list.append(file_dict)?;
88    }
89
90    dict.set_item("files", files_list)?;
91
92    let stats_dict = PyDict::new(py);
93    stats_dict.set_item("total_files", stats.total_files)?;
94    stats_dict.set_item("metadata_ok", stats.metadata_ok)?;
95    stats_dict.set_item("metadata_failed", stats.metadata_failed)?;
96    stats_dict.set_item("hash_ok", stats.hash_ok)?;
97    dict.set_item("stats", stats_dict)?;
98
99    Ok(dict)
100}

123// ─── File Watcher ───────────────────────────────────────────────

125struct WatcherInner {
126    watcher: FileWatcher,
127}
128
129#[pyclass]
130pub struct RustFileWatcher {
131    inner: Arc<Mutex<Option<WatcherInner>>>,
132}
133
134#[pymethods]
135impl RustFileWatcher {
136    #[new]
137    fn new(dirs: Vec<String>) -> PyResult<Self> {
138        let watcher = FileWatcher::new(dirs)
139            .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))?;
140
141        Ok(Self {
142            inner: Arc::new(Mutex::new(Some(WatcherInner { watcher }))),
143        })
144    }
145
146    #[pyo3(signature = (timeout_ms=1000, debounce_ms=2000))]
147    fn poll_changes<'py>(
148        &self,
149        py: Python<'py>,
150        timeout_ms: u64,
151        debounce_ms: u64,
152    ) -> PyResult<pyo3::Bound<'py, PyList>> {
153        let mut guard = self.inner.lock().unwrap();
154        let inner = guard.as_mut().ok_or_else(|| {
155            pyo3::exceptions::PyRuntimeError::new_err("watcher stopped")
156        })?;
157
158        let changes = py.allow_threads(|| {
159            inner.watcher.poll_debounced(
160                Duration::from_millis(timeout_ms),
161                Duration::from_millis(debounce_ms),
162            )
163        });
164
165        let list = PyList::empty(py);
166        for change in &changes {
167            let dict = PyDict::new(py);
168            dict.set_item("path", &change.path)?;
169            dict.set_item("type", match change.change_type {
170                ChangeType::Added => "added",
171                ChangeType::Modified => "modified",
172                ChangeType::Deleted => "deleted",
173            })?;
174            list.append(dict)?;
175        }
176        Ok(list)
177    }
178
179    fn stop(&self) -> PyResult<()> {
180        let mut guard = self.inner.lock().unwrap();
181        if let Some(mut inner) = guard.take() {
182            inner.watcher.stop();
183        }
184        Ok(())
185    }
186}

188pub fn register(m: &pyo3::Bound<'_, pyo3::types::PyModule>) -> PyResult<()> {
189    m.add_function(pyo3::wrap_pyfunction!(list_audio_files, m)?)?;
190    m.add_function(pyo3::wrap_pyfunction!(scan_directories, m)?)?;
191    m.add_function(pyo3::wrap_pyfunction!(audio_hash, m)?)?;
192    m.add_function(pyo3::wrap_pyfunction!(same_quality_tier, m)?)?;
193    m.add_function(pyo3::wrap_pyfunction!(quality_suffix_fn, m)?)?;
194    m.add_class::<RustFileWatcher>()?;
195    Ok(())
