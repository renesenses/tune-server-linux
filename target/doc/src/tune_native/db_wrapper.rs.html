10fn artist_to_dict<'py>(py: Python<'py>, a: &Artist) -> PyResult<pyo3::Bound<'py, PyDict>> {
11    let d = PyDict::new(py);
12    if let Some(id) = a.id { d.set_item("id", id)?; }
13    d.set_item("name", &a.name)?;
14    if let Some(ref v) = a.sort_name { d.set_item("sort_name", v)?; }
15    if let Some(ref v) = a.musicbrainz_id { d.set_item("musicbrainz_id", v)?; }
16    if let Some(ref v) = a.bio { d.set_item("bio", v)?; }
17    if let Some(ref v) = a.image_path { d.set_item("image_path", v)?; }
18    Ok(d)
19}

21fn track_to_dict<'py>(py: Python<'py>, t: &Track) -> PyResult<pyo3::Bound<'py, PyDict>> {
22    let d = PyDict::new(py);
23    if let Some(id) = t.id { d.set_item("id", id)?; }
24    d.set_item("title", &t.title)?;
25    if let Some(v) = t.album_id { d.set_item("album_id", v)?; }
26    if let Some(ref v) = t.album_title { d.set_item("album_title", v)?; }
27    if let Some(v) = t.artist_id { d.set_item("artist_id", v)?; }
28    if let Some(ref v) = t.artist_name { d.set_item("artist_name", v)?; }
29    d.set_item("disc_number", t.disc_number)?;
30    d.set_item("track_number", t.track_number)?;
31    d.set_item("duration_ms", t.duration_ms)?;
32    if let Some(ref v) = t.file_path { d.set_item("file_path", v)?; }
33    if let Some(ref v) = t.format { d.set_item("format", v)?; }
34    if let Some(v) = t.sample_rate { d.set_item("sample_rate", v)?; }
35    if let Some(v) = t.bit_depth { d.set_item("bit_depth", v)?; }
36    d.set_item("channels", t.channels)?;
37    if let Some(ref v) = t.audio_hash { d.set_item("audio_hash", v)?; }
38    d.set_item("source", &t.source)?;
39    if let Some(ref v) = t.genre { d.set_item("genre", v)?; }
40    if let Some(ref v) = t.composer { d.set_item("composer", v)?; }
41    if let Some(ref v) = t.musicbrainz_recording_id { d.set_item("musicbrainz_recording_id", v)?; }
42    Ok(d)
43}

45#[pyclass]
46pub struct RustDatabase {
47    db: SqliteDb,
48}
49
50#[pymethods]
51impl RustDatabase {
52    #[new]
53    fn new(path: &str) -> PyResult<Self> {
54        let db = if path == ":memory:" {
55            SqliteDb::open_in_memory()
56        } else {
57            SqliteDb::open(path)
58        }.map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))?;
59        db.init_schema().map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))?;
60        Ok(Self { db })
61    }
62
63    // ── Artist ──────────────────────────────────────────────
64
65    fn artist_get<'py>(&self, py: Python<'py>, id: i64) -> PyResult<Option<pyo3::Bound<'py, PyDict>>> {
66        let repo = ArtistRepo::new(self.db.clone());
67        match repo.get(id).map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))? {
68            Some(a) => Ok(Some(artist_to_dict(py, &a)?)),
69            None => Ok(None),
70        }
71    }
72
73    fn artist_get_by_name<'py>(&self, py: Python<'py>, name: &str) -> PyResult<Option<pyo3::Bound<'py, PyDict>>> {
74        let repo = ArtistRepo::new(self.db.clone());
75        match repo.get_by_name(name).map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))? {
76            Some(a) => Ok(Some(artist_to_dict(py, &a)?)),
77            None => Ok(None),
78        }
79    }
80
81    fn artist_get_or_create<'py>(&self, py: Python<'py>, name: &str, musicbrainz_id: Option<&str>, sort_name: Option<&str>) -> PyResult<pyo3::Bound<'py, PyDict>> {
82        let repo = ArtistRepo::new(self.db.clone());
83        let a = repo.get_or_create(name, musicbrainz_id, sort_name)
84            .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))?;
85        artist_to_dict(py, &a)
86    }
87
88    fn artist_count(&self) -> PyResult<i64> {
89        let repo = ArtistRepo::new(self.db.clone());
90        repo.count().map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))
91    }
92
93    fn artist_search<'py>(&self, py: Python<'py>, query: &str, limit: i64) -> PyResult<Vec<Py<PyAny>>> {
94        let repo = ArtistRepo::new(self.db.clone());
95        let artists = repo.search(query, limit)
96            .map_err(|e| pyo3::exceptions::PyRuntimeError::new_err(e))?;
97        artists.iter().map(|a| Ok(artist_to_dict(py, a)?.unbind().into_any())).collect()
98    }
