15#[pyfunction]
16fn read_metadata(py: Python<'_>, path: &str) -> PyResult<Option<Py<PyAny>>> {
17    let meta = tune_core::metadata::read_metadata(Path::new(path));
18    match meta {
19        Some(m) => {
20            let dict = PyDict::new(py);
21            macro_rules! set_opt {
22                ($key:expr, $val:expr) => {
23                    if let Some(ref v) = $val { dict.set_item($key, v)?; }
24                };
25            }
26            set_opt!("title", m.title);
27            set_opt!("artist", m.artist);
28            set_opt!("album", m.album);
29            set_opt!("album_artist", m.album_artist);
30            set_opt!("album_artist_sort", m.album_artist_sort);
31            set_opt!("track_number", m.track_number);
32            set_opt!("disc_number", m.disc_number);
33            set_opt!("disc_subtitle", m.disc_subtitle);
34            set_opt!("year", m.year);
35            set_opt!("original_year", m.original_year);
36            set_opt!("release_date", m.release_date);
37            set_opt!("original_date", m.original_date);
38            set_opt!("genre", m.genre);
39            set_opt!("duration_ms", m.duration_ms);
40            set_opt!("sample_rate", m.sample_rate);
41            set_opt!("bit_depth", m.bit_depth);
42            set_opt!("channels", m.channels);
43            set_opt!("format", m.format);
44            set_opt!("file_size", m.file_size);
45            set_opt!("bpm", m.bpm);
46            set_opt!("label", m.label);
47            set_opt!("catalog_number", m.catalog_number);
48            set_opt!("musicbrainz_recording_id", m.musicbrainz_recording_id);
49            set_opt!("musicbrainz_release_id", m.musicbrainz_release_id);
50            set_opt!("musicbrainz_artist_id", m.musicbrainz_artist_id);
51            set_opt!("musicbrainz_album_artist_id", m.musicbrainz_album_artist_id);
52            set_opt!("musicbrainz_release_group_id", m.musicbrainz_release_group_id);
53            set_opt!("isrc", m.isrc);
54            dict.set_item("has_cover", m.has_cover)?;
55            dict.set_item("compilation", m.compilation)?;
56
57            if !m.credits.is_empty() {
58                let credits = PyList::empty(py);
59                for c in &m.credits {
60                    let cd = PyDict::new(py);
61                    cd.set_item("name", &c.name)?;
62                    cd.set_item("role", &c.role)?;
63                    if let Some(ref inst) = c.instrument {
64                        cd.set_item("instrument", inst)?;
65                    }
66                    credits.append(cd)?;
67                }
68                dict.set_item("credits", credits)?;
69            }
70
71            Ok(Some(dict.unbind().into_any()))
72        }
73        None => Ok(None),
74    }
75}

88#[pyfunction]
89fn format_from_extension(ext: &str) -> Option<String> {
90    tune_core::audio::formats::AudioFormat::from_extension(ext)
91        .map(|f| format!("{:?}", f).to_lowercase())
92}

94#[pyfunction]
95fn mime_type_for_format(format_name: &str) -> String {
96    let fmt = match format_name {
97        "flac" => tune_core::audio::formats::AudioFormat::Flac,
98        "wav" => tune_core::audio::formats::AudioFormat::Wav,
99        "mp3" => tune_core::audio::formats::AudioFormat::Mp3,
100        "aac" => tune_core::audio::formats::AudioFormat::Aac,
101        "alac" => tune_core::audio::formats::AudioFormat::Alac,
102        "ogg" => tune_core::audio::formats::AudioFormat::Ogg,
103        "opus" => tune_core::audio::formats::AudioFormat::Opus,
104        "aiff" => tune_core::audio::formats::AudioFormat::Aiff,
105        "dsd" => tune_core::audio::formats::AudioFormat::Dsd,
106        "wavpack" => tune_core::audio::formats::AudioFormat::WavPack,
107        "ape" => tune_core::audio::formats::AudioFormat::Ape,
108        _ => return "application/octet-stream".to_string(),
109    };
110    fmt.mime_type().to_string()
111}

113#[pymodule]
114fn tune_native(m: &Bound<'_, PyModule>) -> PyResult<()> {
115    m.add_function(wrap_pyfunction!(version, m)?)?;
116    m.add_function(wrap_pyfunction!(read_metadata, m)?)?;
117    m.add_function(wrap_pyfunction!(build_wav_header, m)?)?;
118    m.add_function(wrap_pyfunction!(find_ffmpeg, m)?)?;
119    m.add_function(wrap_pyfunction!(format_from_extension, m)?)?;
120    m.add_function(wrap_pyfunction!(mime_type_for_format, m)?)?;
121    m.add_class::<pipeline_wrapper::RustPipeline>()?;
122    m.add_class::<discovery_wrapper::RustSsdpScanner>()?;
123    m.add_class::<discovery_wrapper::RustMdnsScanner>()?;
124    scanner_wrapper::register(m)?;
125    db_wrapper::register(m)?;
126    Ok(())
127}
